SELECT 
    IDOFICINA,
    REPLACE(UPPER(SUBSTR(CODIGOPOSTAL, 1, 1)), 'A', '@') AS CODIGOPOSTAL,
    REPLACE(UPPER(SUBSTR(CIUDAD, 1, 1)), 'A', '@') AS CIUDAD,
    REPLACE(UPPER(SUBSTR(ESTADO, 1, 1)), 'A', '@') AS ESTADO,
    REPLACE(UPPER(SUBSTR(PAIS, 1, 1)), 'A', '@') AS PAIS
FROM (
    SELECT 
        IDOFICINA, DIRECCION,
        NVL2(CODPOSTAL, CODPOSTAL, CONCAT(UPPER(SUBSTR(DIRECCION, 1, 1)), LOWER(SUBSTR(DIRECCION, -1)))) AS CODIGOPOSTAL,
        NVL2(CIUDAD, CIUDAD, CONCAT(UPPER(SUBSTR(DIRECCION, 1, 1)), LOWER(SUBSTR(DIRECCION, -1)))) AS CIUDAD,
        NVL2(ESTADO, ESTADO, CONCAT(UPPER(SUBSTR(DIRECCION, 1, 1)), LOWER(SUBSTR(DIRECCION, -1)))) AS ESTADO,
        NVL2(IDPAIS, IDPAIS, CONCAT(UPPER(SUBSTR(DIRECCION, 1, 1)), LOWER(SUBSTR(DIRECCION, -1)))) AS PAIS
    FROM OFICINAS
)
ORDER BY LENGTH(CODIGOPOSTAL) DESC;

SELECT
    UPPER(SUBSTR(NOMBRE,1,1)) as NOMBRE
FROM EMPLEADOS
WHERE (
    NOMBRE LIKE '%a%'
    OR NOMBRE LIKE '%e%'
)AND
AND SUBSTR(UPPER(NOMBRE), 1, 1) NOT IN ('A','E','I','O','U');

SELECT INITCAP(CONCAT(CONCAT(NOMBRE,' '),APELLIDO)) as NCOMPLETO,
DECODE(MOD(LENGTH(NOMBRE),2), 0 ,'Divisible','No divisible') as DIV,
INITCAP(SUBSTR(NOMBRE,1,1)) as PLN,
INITCAP(SUBSTR(APELLIDO,1,1)) as PLA,
DECODE(IDJEFE, null, 'JEFAZO','NOJEFE') as TJEFE
FROM EMPLEADOS
WHERE UPPER(SUBSTR(NOMBRE, 2,4)) LIKE '%A%';

SELECT 
    INITCAP(NOMBRE || ' ' || APELLIDO) AS EMPLEADO,
    DIAINGRESO,
    TRUNC(MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) AS ANIOS_ANTIGUEDAD,
    CASE 
        WHEN MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 <= 5 THEN 'NOVATO'
        WHEN MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 <= 10 THEN 'EXPERIMENTADO'
        ELSE 'VETERANO'
    END AS CATEGORIA,
    ADD_MONTHS(DIAINGRESO, 12) AS PROXIMO_ANIVERSARIO
FROM EMPLEADOS;
