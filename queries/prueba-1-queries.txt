

DESCRIBE EMPLEADOS;

SELECT *
FROM EMPLEADOS;

1)

SELECT NOMBRECOMPLETO
FROM (
SELECT 
UPPER(SUBSTR(NOMBRE, 1, 1)) || '' || SUBSTR(LOWER(NOMBRE), 2) || ' ' || APELLIDO AS NOMBRECOMPLETO,
NOMBRE,
APELLIDO
FROM EMPLEADOS
WHERE LENGTH(APELLIDO) = 6
)
WHERE 
LOWER(SUBSTR(NOMBRE, 1, 1)) IN ('a', 'e', 'i', 'o', 'u')
AND LOWER(SUBSTR(APELLIDO, 1, 1)) IN ('a', 'e', 'i', 'o', 'u')
AND LOWER(SUBSTR(APELLIDO, -1)) IN ('a', 'e', 'i', 'o', 'u');


2)

SELECT 
INITCAP(NOMBRE) || ' ' || INITCAP(APELLIDO) AS NOMBRECOMPLETO,
CASE 
WHEN LENGTH(NOMBRE) < 6 THEN 'CORTO'
ELSE 'LARGO'
END AS LNOMBRE
FROM EMPLEADOS
WHERE LENGTH(NOMBRE) BETWEEN 5 AND 8;

3)

SELECT 
SUBSTR(NOMBRE, 1, 3) || '-' || SUBSTR(APELLIDO, -2) AS CODIGO_GENERADO
FROM EMPLEADOS
WHERE MOD(LENGTH(NOMBRE) + LENGTH(APELLIDO), 2) <> 0;

4)

SELECT NOMBRE
FROM EMPLEADOS
WHERE (
UPPER(SUBSTR(NOMBRE, 3, 1)) = 'R' 
OR UPPER(SUBSTR(NOMBRE, 6)) LIKE '%S%'
)
AND DIANACIMIENTO BETWEEN TO_DATE('01/01/1980', 'DD/MM/YYYY') 
AND TO_DATE('31/12/1989', 'DD/MM/YYYY');

5)

SELECT 
CONCAT(CONCAT(INITCAP(NOMBRE), ' '), INITCAP(APELLIDO)) AS NOMBRE_COMPLETO,
INSTR(LOWER(NOMBRE), 'a') AS POSICION_PRIMERA_A,
CASE 
WHEN MOD(LENGTH(NOMBRE), 2) = 0 THEN 'PAR'
ELSE 'IMPAR'
END AS LONGITUD_NOMBRE,
UPPER(SUBSTR(APELLIDO, -3)) AS ULTIMAS_3_APELLIDO
FROM EMPLEADOS
WHERE MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 > 20;


6)

DESCRIBE DEPARTAMENTOS;

SELECT *
FROM DEPARTAMENTOS;


SELECT *
FROM DEPARTAMENTOS
WHERE MOD(SUBSTR(IDDEP,1,1),2) <> 0;

SELECT 
NOMBRE AS ORIGINAL,
CONCAT(
SUBSTR(nombre, 1, CEIL(LENGTH(nombre)/2) - 1),
CONCAT('$', SUBSTR(nombre, CEIL(LENGTH(nombre)/2) + 1))
) AS MODIFICADO,
IDDEP AS ID
FROM DEPARTAMENTOS
WHERE NOT MOD(SUBSTR(TO_CHAR(IDDEP), 1, 1), 3) <> 0;


7)

SELECT 
SUBSTR(NOMBRE,1,1) || SUBSTR(APELLIDO,-1) || LENGTH(NOMBRE || ' ' || APELLIDO) AS CODIGO
FROM EMPLEADOS
WHERE 
(
UPPER(SUBSTR(NOMBRE,1,1)) NOT IN ('A','E','I','O','U')
AND 
UPPER(SUBSTR(NOMBRE,-1)) NOT IN ('A','E','I','O','U')
)
AND CEIL(MONTHS_BETWEEN(SYSDATE, DIANACIMIENTO)/12) >= 65
AND CEIL(MONTHS_BETWEEN(SYSDATE, DIAINGRESO)/12) >= 25;


8)

SELECT 
DIRECCION,
CASE 
WHEN ESTADO IS NULL THEN 'NO ESPECIFICADO'
ELSE SUBSTR(ESTADO, 1, 4)
END AS ESTADO_INFO,
NVL(CODPOSTAL, 'SIN CODIGO') AS CODIGO_POS,
CASE 
WHEN ESTADO IS NOT NULL AND CODPOSTAL IS NOT NULL THEN 'COMPLETA'
ELSE 'INCOMPLETA'
END AS COMPLETITU
FROM OFICINAS;

9)

SELECT 
INITCAP(NOMBRE || ' ' || APELLIDO) AS "Nombre Completo",
TRUNC(MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) AS "Antiguedad",
SALARIO,
CASE 
WHEN MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 > 25 AND SALARIO > 2000000 THEN 'ALTA GAMA'
WHEN MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 BETWEEN 15 AND 25 
AND SALARIO BETWEEN 1000000 AND 2000000 THEN 'MEDIA GAMA'
WHEN MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12 < 15 THEN 'BAJA GAMA'
ELSE 'OTRA' 
END AS "Categoría GAMA",
DECODE(IDJEFE, NULL, 'No tiene jefe', 'Tiene jefe') AS "Situación Jefatura"
FROM empleados;

10)

SELECT 
INITCAP(NOMBRE || ' ' || APELLIDO) AS "NOMBRECOMPLETO",
TRUNC(MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) AS "ANTIGUEDAD",
SALARIO,
CASE 
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) > 25 AND SALARIO > 2000000 
THEN 'ALTA GAMA'
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) BETWEEN 15 AND 25 
AND SALARIO BETWEEN 1000000 AND 2000000 
THEN 'MEDIA GAMA'
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) < 15 
THEN 'BAJA GAMA'
ELSE 'FUERA DE RANGO'
END AS "CATEGORIAGAMA",
DECODE(IDJEFE, NULL, 'No tiene jefe', 'Tiene jefe') AS "SJEFATURA"
FROM EMPLEADOS;

11)

SELECT 
INITCAP(NOMBRE) AS NOMBRE,
NVL(IDCONT, 0) AS ID,
DECODE(
IDCONT, 
1, 'EUROPA', 
2, 'AMERICAS', 
3, 'ASIA', 
4, 'AFRICA', 
'OTROS'
) AS CONTINENTE,
CASE 
WHEN IDCONT IN (1, 3, 4) THEN 'Hemisferio ESTE'
ELSE 'OESTE'
END AS "Indicador Hemisferio"
FROM PAISES;

12)

SELECT 
INITCAP(NOMBRE || ' ' || APELLIDO) AS "Empleado",
TRUNC(MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) AS "Antiguedad",
SALARIO,
CASE 
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) > 20 AND SALARIO > 1000000 
THEN 'SENIOR ALTO'
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) > 20 AND SALARIO BETWEEN 500000 AND 1000000 
THEN 'SENIOR MEDIO'
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) BETWEEN 10 AND 20 
THEN 'JUNIOR AVANZADO'
WHEN (MONTHS_BETWEEN(SYSDATE, DIAINGRESO) / 12) < 10 
THEN 'JUNIOR'
ELSE 'FUERA DE RANGO'
END AS "Categoría",
NVL2(IDJEFE, 'Tiene jefe', 'No tiene jefe') AS "Situación Jefatura"
FROM EMPLEADOS;

13)

SELECT
idpuesto,
INITCAP(titulo) AS titulo_formateado,
salario_min,
salario_max,
CASE
WHEN salario_max > 2000000 THEN 'ALTO'
WHEN salario_max BETWEEN 1000000 AND 2000000 THEN 'MEDIO'
ELSE 'BAJO'
END AS rango_salarial,
(salario_max - salario_min) AS diferencia_salarial,
CASE
WHEN (salario_max - salario_min) > 1000000 THEN 'AMPLIO'
ELSE 'ESTRECHA'
END AS amplitud,
DECODE(SUBSTR(idpuesto,1,1),
'A', 'ADMINISTRATIVO',
'B', 'OPERATIVO',
'C', 'TECNICO',
'D', 'ASISTENCIAL',
'OTRO'
) AS categoria
FROM puestos;
